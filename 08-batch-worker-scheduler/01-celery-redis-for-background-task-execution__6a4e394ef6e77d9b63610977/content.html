# Typical Producer Flow in FastAPI
from gmsshared.celery_utils import celery_app

@router.post("/signup")
def member_signup(member_data: MemberSchema):
    # 1. Save to DB (Fast)
    # 2. Hand off the slow stuff
    celery_app.send_task(
        "owner_verification_email", 
        args=[member_data.email], 
        queue="prod-gms:emails"
    )
    return {"status": "Processing"} # Respond immediately
