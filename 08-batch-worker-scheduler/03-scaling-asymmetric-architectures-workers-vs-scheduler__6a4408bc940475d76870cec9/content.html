@celery_app.task(name="batch.credit_noshows")
def credit_noshows():
    # Use a managed session scope for atomicity
    with db.session_scope() as session:
        expired_bookings = session.query(Booking).filter(
            Booking.status == 'booked',
            Booking.class_time < datetime.utcnow()
        ).all()
        
        for booking in expired_bookings:
            # Domain logic for credit deduction
            process_member_penalty(session, booking.member_id)
            booking.status = 'processed_noshow'
