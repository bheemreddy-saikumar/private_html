Implementation and Integration

Build shared logic into the base; microservices stay thin and consistent.

gms-base Dockerfile

# gms-base.Dockerfile
FROM python:3.11-slim-bullseye

WORKDIR /app

# Install system-level dependencies common to all services
RUN apt-get update && apt-get install -y \
    build-essential \
    default-libmysqlclient-dev \
    pkg-config \
    && rm -rf /var/lib/apt/lists/*

# Install shared internal library (gmsshared)
COPY ./gmsshared /app/gmsshared
RUN pip install --no-cache-dir ./gmsshared

Thin Service Dockerfile

# auth-admin-service/Dockerfile
FROM gms-base:latest

# Copy only service-specific requirements
COPY ./auth-admin-service/requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

COPY ./auth-admin-service /app/auth-admin-service

# Execution context
ENV PYTHONPATH=/app
CMD ["uvicorn", "auth-admin-service.src.main:app", "--host", "0.0.0.0", "--port", "8000"]
