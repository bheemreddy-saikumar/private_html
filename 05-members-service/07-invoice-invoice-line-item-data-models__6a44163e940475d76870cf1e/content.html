@property
def total_amount(self):
    return sum([item.total_amount for item in self.invoice_items 
                if item.invoice_item_status_type_id != InvoiceItemStatusTypeEnum.CANCELLED.value])

Dynamic Due Date

# From gms_be/gmsshared/src/models/invoice.py
@property
def due_date(self):
    items_with_due_date = [item for item in self.invoice_items if item.due_date is not None]
    if not items_with_due_date:
        return None
    return max(items_with_due_date, key=lambda item: item.due_date).due_date
