# gmsshared/src/web/fastapi_glue.py
def _check_location_access(user: User, request: Request) -> None:
    # Logic to ensure the user actually belongs to the gym_id or location_id
    # in the request. This prevents a manager at Gym A from seeing members at Gym B.
    gym_id = request.query_params.get("gym_id")
    if gym_id and int(gym_id) not in [loc.gym.id for loc in user.locations]:
        raise ApiForbidden(f"You are not authorized to access Gym id: {gym_id}")
