from sqlalchemy import create_engine
from gmsshared.src.config import get_config

engine = create_engine(
    get_config().SQLALCHEMY_DATABASE_URI,
    pool_pre_ping=True,     # Check if connection is alive before handing it out
    pool_recycle=3600,      # Drop connections older than 1 hour
    future=True,

session_scope() Context Manager

# gmsshared/src/database.py
@contextmanager
def session_scope():
    """Session context for non-request code (Celery tasks, scripts)."""
    try:
        yield session
        session.commit()
    except Exception:
        session.rollback()
        raise
    finally:
        session.remove()
