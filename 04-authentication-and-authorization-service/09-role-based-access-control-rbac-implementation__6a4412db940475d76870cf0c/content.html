Four sequential gates validate every protected request.

check_access Pipeline

def check_access(roles=None, return_role=False, return_user_id=False):
    def decorate(fn):
        @wraps(fn)
        def decorated(*args, **kwargs):
            # 1. Identity Validation (Authentication)
            token_payload = _check_access_token()
            
            # 2. Database Integrity Check
            user = _check_user(int(token_payload['user_id']))
            
            # 3. RBAC Validation (The Role Check)
            _check_roles(user, roles)
            
            # 4. Contextual Validation (Multi-tenancy/Location Check)
            _check_location_access(user)
