from pydantic import BaseModel, EmailStr, Field
from uuid import UUID

class AdminUserCreateRequest(BaseModel):
    email: EmailStr
    role_id: int = Field(..., gt=0)
    gym_id: UUID
    first_name: str = Field(..., min_length=1, max_length=50)
    is_active: bool = True
