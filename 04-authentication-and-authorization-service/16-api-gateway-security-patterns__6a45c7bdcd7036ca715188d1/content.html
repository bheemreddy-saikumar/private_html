# Conceptual logic for a Gateway Rate Limiter (e.g., using Redis)
def is_rate_limited(client_id: str, limit: int, window: int) -> bool:
    current_usage = redis.get(f"rate_limit:{client_id}")
    if current_usage and int(current_usage) >= limit:
        return True
    
    pipe = redis.pipeline()
    pipe.incr(f"rate_limit:{client_id}")
    pipe.expire(f"rate_limit:{client_id}", window)
    pipe.execute()
    return False
